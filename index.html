import time import pandas as pd from geopy.geocoders import Nominatim from geopy.extra.rate_limiter import RateLimiter # 1. Dateipfade festlegen EINGABE_CSV = "restaurants_1896.csv" # Name deiner Eingabedatei AUSGABE_CSV = "restaurants_1896_geocoded.csv" # 2. CSV einlesen (trennt nach Komma oder Semikolon automatisch) df = pd.read_csv(EINGABE_CSV, sep=None, engine="python") # 3. Geocoder initialisieren (Einen eigenen user_agent wählen!) geolocator = Nominatim(user_agent="posen_historical_mapping_project_v1") # RateLimiter verhindert, dass Nominatim deine IP wegen zu vieler Anfragen blockiert (min 1 Sekunde Pause) geocode = RateLimiter(geolocator.geocode, min_delay_seconds=1.0) # 4. Such-Adresse vorbereiten (Fügt "Poznań, Poland" an, falls nicht vorhanden) def prepare_address(raw_address): if pd.isna(raw_address) or not str(raw_address).strip(): return None addr = str(raw_address).strip() if "Poznań" not in addr and "Poznan" not in addr: addr += ", Poznań, Poland" return addr print("Starte Geocoding...") # 5. Geocoding-Funktion für jede Zeile def get_coordinates(row): # Falls bereits Koordinaten vorhanden sind, überspringen if pd.notna(row.get("lat")) and pd.notna(row.get("lng")): return row.get("lat"), row.get("lng") address = prepare_address(row.get("Polish address")) if not address: print(f"[-] Keine Adresse für: {row.get('name')}") return None, None try: location = geocode(address) if location: print( f"[✓] Gefunden: {row.get('name')} ({address}) -> {location.latitude}, {location.longitude}" ) return location.latitude, location.longitude else: print(f"[x] Nicht gefunden: {address}") return None, None except Exception as e: print(f"[!] Fehler bei {address}: {e}") return None, None # 6. Koordinaten abrufen und Spalten befüllen coords = df.apply(get_coordinates, axis=1) df["lat"] = [c[0] for c in coords] df["lng"] = [c[1] for c in coords] # 7. Ergebnis als neue CSV speichern df.to_csv(AUSGABE_CSV, index=False, encoding="utf-8") print(f"\nFertig! Die geocodierte Datei wurde als '{AUSGABE_CSV}' gespeichert.")
